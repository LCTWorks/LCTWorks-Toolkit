using LCTWorks.Web;
using LCTWorks.Web.Tools;

// 1. Link preview (providers → HTML crawl → fallbacks), cached.
var preview = await WebPreviewer.Default.GetPreviewAsync("https://www.youtube.com/watch?v=dQw4w9WgXcQ", cancellationToken);

if (preview.IsBlocked)
{
    // Only site name + icon are available.
    Debug.WriteLine($"Blocked: {preview.BlockReason}");
}

Title = preview.Title;
Description = preview.Description;
SiteName = preview.SiteName;
ThumbnailSource = preview.ImageUrl;
IconSource = preview.IconUrl;

// 2. Custom configuration.
// Hidden WebView2 (real Edge) for pages behind automatic bot checks, e.g. Cloudflare's "Just a moment...".
// FetcherWebView must be in the visual tree (1x1, Opacity="0").
var webViewFetcher = new WebView2PageFetcher(FetcherWebView);

var options = new WebPreviewerOptions
{
    RetryAsPreviewBot = true,
    CacheDuration = TimeSpan.FromHours(1),
    FallbackIconUrlTemplate = null, // don't send domains to Google's favicon service
    BlockedPageFetcher = webViewFetcher.FetchHtmlAsync,
};

// Your own site-specific provider runs before the built-in ones.
options.Providers.Insert(0, new MyIntranetPreviewProvider());

var previewer = new WebPreviewer(options);
var custom = await previewer.GetPreviewAsync("stackoverflow.com/questions/11227809", cancellationToken);
Debug.WriteLine(webViewFetcher.LastResult?.Note); // e.g. "Passed bot challenge"

// 3. Raw crawler: meta tags, JSON-LD, images.
var crawler = await WebCrawler.FromUrlAsync("https://github.com/dotnet/runtime", cancellationToken);
if (crawler.IsLoaded)
{
    var title = crawler.GetTitle();
    var image = crawler.GetPreviewImage();
    var icon = crawler.GetIconUrl();
    var oembed = crawler.GetOEmbedUrl();
    var themeColor = crawler.GetMeta("theme-color");
    var meta = crawler.GetMetaTags();
    var images = await crawler.GetAllImagesAsync([".svg", ".ico"], validateImages: true, cancellationToken);
}

// Custom provider.
public sealed class MyIntranetPreviewProvider : IPreviewProvider
{
    public string Name => "Intranet";

    public bool IsAuthoritative => true; // skip the generic crawl when this returns a result

    public bool CanHandle(Uri uri) => uri.Host.EndsWith(".contoso.local", StringComparison.OrdinalIgnoreCase);

    public async Task<WebPreview?> GetPreviewAsync(Uri uri, HttpClient client, CancellationToken cancellationToken)
    {
        var doc = await intranetApi.GetDocumentAsync(uri, cancellationToken);
        return doc == null ? null : new WebPreview
        {
            Url = uri.AbsoluteUri,
            Title = doc.Title,
            Description = doc.Summary,
            ImageUrl = doc.ThumbnailUrl,
            SiteName = "Contoso Intranet",
            Provider = Name,
        };
    }
}
